cmake_minimum_required(VERSION 3.16)
project(Frigo VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(QT NAMES Qt6 Qt5 REQUIRED COMPONENTS Widgets SerialBus Network Test)
find_package(Qt${QT_VERSION_MAJOR} REQUIRED COMPONENTS Widgets SerialBus Network Test)
add_library(frigo_core src/protocol.cpp src/controller.cpp)
target_include_directories(frigo_core PUBLIC src)
target_link_libraries(frigo_core PUBLIC Qt${QT_VERSION_MAJOR}::SerialBus Qt${QT_VERSION_MAJOR}::Network)
add_executable(frigo src/main.cpp src/window.cpp src/network.cpp assets/assets.qrc)
target_link_libraries(frigo PRIVATE frigo_core Qt${QT_VERSION_MAJOR}::Widgets)
if(MINGW)
 target_link_options(frigo PRIVATE -mwindows)
endif()
enable_testing()
add_executable(protocol_tests tests/protocol_tests.cpp)
target_link_libraries(protocol_tests PRIVATE frigo_core Qt${QT_VERSION_MAJOR}::Test)
add_test(NAME protocol COMMAND protocol_tests)
add_executable(window_tests tests/window_tests.cpp src/window.cpp src/network.cpp assets/assets.qrc)
target_link_libraries(window_tests PRIVATE frigo_core Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Test)
add_test(NAME interface COMMAND window_tests -platform offscreen -o -,txt)
install(TARGETS frigo RUNTIME DESTINATION bin)
